# WP1: plant_equations.py — 8 полных ОДУ с переменной уставкой g(t)

## Назначение
Создать `plant_equations.py` с классом `PlantODESystem`, реализующим **8 физических
дифференциальных уравнений** для 8 компонент объекта управления участка металлообработки:
токарная обработка (x1, x2, x3), фрезерная обработка (x4, x5, x6), закалка (x7, x8).

Текущая модель в `plant.py` описывает каждую компоненту как **периодическое звено**
(простое апериодическое звено первого порядка с постоянной времени). Это упрощение
недопустимо: нужны **полноценные дифференциальные уравнения технологического процесса**
с учётом физики каждой операции.

Уставка `g(t)` — **функция времени**, а не константа. Она отражает программу обработки:
нагрев → выдержка → охлаждение при закалке; плавно меняющийся профиль при точении
и фрезеровании. Для каждой компоненты задаётся своя допустимая полоса `±δ_i`.

import numpy as np

class PlantODESystem:
    N_STATES = 8

    def __init__(self, params: dict | None = None):
        """params — переопределение параметров по умолчанию (словарь {имя: значение})."""

    def initial_state(self) -> np.ndarray:
        """Начальное состояние — центры контрольных карт (или близкие к ним), shape (8,)."""

    def setpoint(self, t: float) -> np.ndarray:
        """Уставка g(t) — функция времени, shape (8,). НЕ константа."""

    def tolerance(self) -> np.ndarray:
        """Допустимая полоса ±δ_i для каждого показателя, shape (8,), все значения > 0."""

    def derivatives(self, x: np.ndarray, u: np.ndarray, t: float) -> np.ndarray:
        """Правые части ОДУ dx/dt, shape (8,)."""

    def step(self, x: np.ndarray, u: np.ndarray, dt: float, t: float,
             noise: np.ndarray | None = None,
             disturbance: np.ndarray | None = None) -> np.ndarray:
        """Один шаг интегрирования (метод Эйлера или Рунге-Кутты). shape (8,).
        noise — аддитивный шум к производной, disturbance — детерминированное возмущение
        (тренд износа, скачок вибрации, недостаточное охлаждение и т.п.)."""

    def equilibrium(self) -> tuple[np.ndarray, np.ndarray]:
        """Точка равновесия (x*, u*), в которой derivatives(x*, u*, t) ≈ 0 для типичного t.
        Если равновесие зависит от t (из-за g(t)), вернуть средние значения на горизонте [0, T]."""
